\section*{Explainable AI} %%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Interpretability}
``The ability to explain or to present in understandable terms to humans''\\
\bigskip
Manifestations (Doshi-Velez and Kim 17)
 \begin{itemize}
  \item Problem formulation
  \item Evaluation level?
  \item Task-related\\
  (e.g. global/local, incompleteness, user expertise, time)
  \item Method-related\\
  (e.g. form/number/compositionality of cognitive chunks, monotonicity, uncertainty)
 \end{itemize}
\end{frame}

\begin{frame}{Explanation Methods}
 \begin{tabular}{>{\raggedright}p{0.4\textwidth}>{\raggedright}p{0.4\textwidth}}
  Perturbation
  \begin{itemize}
   \item Meaningful Perturbations\\(Fong \& Vedaldi 17)
  \end{itemize}
   & Function-Based
  \begin{itemize}
   \item Taylor Decomposition
   \item Symbolic Regression
  \end{itemize}\tabularnewline
  Surrogate models
  \begin{itemize}
   \item SmoothGrad\\(Smilkov et al. 16)
  \end{itemize}
   & Structure Based
  \begin{itemize}
   \item Deep Taylor Decomposition\\(Montavon et al. 17)
   \item LRP (Bach et al. 15)
  \end{itemize}\tabularnewline
 \end{tabular}
\end{frame}
