\begin{frame}{Interpretability}
``The ability to explain or to present in understandable terms to humans''\\
\bigskip
Manifestations (Doshi-Velez and Kim 17)
 \begin{itemize}
  \item Problem formulation
  \item Evaluation level
  \item Task-related (e.g. global/local, incompleteness, user expertise, time)
  \item Method-related (e.g. form/number/compositionality of cognitive chunks, monotonicity, uncertainty)
 \end{itemize}
\end{frame}

\begin{frame}{Mountain Car Expert: Parabolic Bounds}
  \begin{algorithm}[H]
 	\SetAlgoLined
 	low $\leftarrow \min(-0.09(\text{pos}+0.25)^2+0.03,0.3(\text{pos}+0.09)^4-0.008)$\;  
 	upper $\leftarrow -0.07(\text{pos}+0.38)^2+0.07)$\;  
 	\uIf{low  $\leq \text{vel} \leq$ upper}
 	{left\;}
 	\Else{right\;}
 	\caption{Parabolic mountain car formula}
  \end{algorithm}
\end{frame}

\begin{frame}{Mountain Car Expert: Nested If-Statements}
  \begin{algorithm}[H]
	\SetAlgoLined
	\uIf{pos $< -1$ or (pos $<$ 0.1 and vel < $-0.05$)}
	{right\;}  
	\uElseIf{vel $< 0.2$ and (pos $> -0.45$ and pos $< -0.05$)}  
	{left\;}  
	\Else{  
		\Comment{The simple known solution.}  
		\uIf{vel $< 0$}
		{left\;}
		\Else{right\;}
	}
	\caption{Nested if-statements to define reversal points.}
  \end{algorithm}
\end{frame}
